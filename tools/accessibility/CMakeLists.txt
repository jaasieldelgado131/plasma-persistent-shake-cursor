cmake_minimum_required(VERSION 3.16)
project(AccessibilityKeepCursorSize VERSION 6.8.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ECM 6.30 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core DBus Quick Widgets)
find_package(Qt6GuiPrivate 6.11 REQUIRED NO_MODULE)
find_package(KF6 6.30 REQUIRED COMPONENTS Config CoreAddons KCMUtils GlobalAccel GuiAddons I18n NotifyConfig XmlGui)
# Match the parent kcms/CMakeLists.txt in plasma-desktop.
remove_definitions(-DQT_NO_CAST_FROM_ASCII -DQT_STRICT_ITERATORS -DQT_NO_CAST_FROM_BYTEARRAY -DQT_NO_KEYWORDS)
if(NOT DEFINED PLASMA_DESKTOP_SOURCE_DIR)
    message(FATAL_ERROR "Set PLASMA_DESKTOP_SOURCE_DIR to the patched plasma-desktop source tree")
endif()
add_subdirectory("${PLASMA_DESKTOP_SOURCE_DIR}/kcms/access" access)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(accessibility_check accessibility_check.cpp)
target_link_libraries(accessibility_check PRIVATE Qt6::Widgets Qt6::Quick Qt6::DBus Qt6::Test KF6::KCMUtilsQuick KF6::ConfigCore KF6::I18n)
